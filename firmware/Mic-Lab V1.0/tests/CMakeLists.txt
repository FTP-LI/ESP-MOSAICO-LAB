cmake_minimum_required(VERSION 3.24)
project(mic_ui_check C)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("../managed_components/lvgl__lvgl" lvgl)
add_executable(ui_check preview_ui.c ../main/mic_font.c)
target_include_directories(ui_check PRIVATE . ../main)
target_compile_options(ui_check PRIVATE -Wall -Wextra)
target_link_libraries(ui_check PRIVATE lvgl m)

cmake_minimum_required(VERSION 3.24)
project(trader_ui_check C)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("../managed_components/lvgl__lvgl" lvgl)
set(JSON "${CMAKE_CURRENT_SOURCE_DIR}/../managed_components/espressif__cjson/cJSON")
add_executable(ui_check preview_ui.c ../main/model.c ../main/watchlist_seed.c ../main/global_market.c ../main/gbk.c ../main/trader_font.c "${JSON}/cJSON.c")
target_include_directories(ui_check PRIVATE . ../main "${JSON}")
target_compile_options(ui_check PRIVATE -Wall -Wextra -Wno-unused-function -Wno-unused-variable)
target_compile_definitions(ui_check PRIVATE TL_HOST)
target_link_libraries(ui_check PRIVATE lvgl m)

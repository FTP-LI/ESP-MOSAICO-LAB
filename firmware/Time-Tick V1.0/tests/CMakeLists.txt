cmake_minimum_required(VERSION 3.24)
project(time_tick_tests C)
enable_testing()
set(TT_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../main/core")
add_executable(test_core test_core.c "${TT_CORE}/timetick_core.c")
add_executable(test_settings test_settings.c "${TT_CORE}/settings_model.c" "${TT_CORE}/timetick_core.c" "${TT_CORE}/alarm_model.c")
add_executable(test_alarm test_alarm.c "${TT_CORE}/alarm_model.c")
add_executable(test_pet test_pet.c "${TT_CORE}/pet_model.c")
add_executable(test_animation test_animation.c "${TT_CORE}/pet_animation.c")
add_executable(test_power test_power.c "${TT_CORE}/power_model.c")
add_executable(test_weather test_weather.c "${TT_CORE}/weather_model.c")
add_executable(test_scene test_scene.c "${TT_CORE}/scene_model.c")
set(TT_LOCATION_CJSON "${CMAKE_CURRENT_SOURCE_DIR}/../managed_components/espressif__cjson/cJSON")
add_executable(test_location test_location.c "${TT_CORE}/location_model.c" "${TT_LOCATION_CJSON}/cJSON.c")
target_include_directories(test_location PRIVATE "${TT_LOCATION_CJSON}")
foreach(test IN ITEMS core settings alarm pet animation power weather scene location)
    target_include_directories(test_${test} PRIVATE "${TT_CORE}")
    if(MSVC)
        target_compile_options(test_${test} PRIVATE /W4 /WX)
    else()
        target_compile_options(test_${test} PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(test_${test} m)
    endif()
    add_test(NAME ${test} COMMAND test_${test})
endforeach()
